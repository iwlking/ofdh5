# ofdh5 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付与 pdfh5 同体验的 OFD 移动端预览库：`@ofdh5/core` 内核 + `@ofdh5/vue2` 组件。

**Architecture:** 解析层把 OFD 的 zip+XML 归一化为与格式无关的 IR（`Page`/`TextItem`/`PathItem`/`ImageItem`），绘制层只吃 IR 输出 canvas，交互层负责手势与滚动，`Ofdh5` 门面编排三者并暴露 pdfh5 风格 API。解析层不认识 canvas，绘制层不认识 zip，因此两层可独立测试。

**Tech Stack:** TypeScript 5.6、fflate 0.8（唯一运行时依赖）、Vite 5（lib 构建 ESM+UMD）、Vitest 2 + jsdom、Playwright、Vue 2.6。

**Spec:** `docs/superpowers/specs/2026-10-23-ofdh5-design.md`

```ts
export type OfdErrorCode =
  | 'E_NETWORK'
  | 'E_NOT_ZIP'
  | 'E_NO_ENTRY'
  | 'E_PARSE_XML'
  | 'E_ENCRYPTED'
  | 'E_FONT'
  | 'E_IMAGE'
  | 'E_CANVAS'
  | 'E_RENDER';

export class OfdError extends Error {
  readonly code: OfdErrorCode;
  constructor(code: OfdErrorCode, message: string) {
    super(message);
    this.name = 'OfdError';
    this.code = code;
  }
}

export function isOfdError(value: unknown): value is OfdError {
  return value instanceof OfdError;
}

export function toOfdError(value: unknown, fallback: OfdErrorCode): OfdError {
  if (isOfdError(value)) return value;
  const message = value instanceof Error ? value.message : String(value);
  return new OfdError(fallback, message);
}
```

```ts
// packages/core/src/events.ts
type HandlerMap = Record<string, (...args: any[]) => void>;

export class EventBus<T extends HandlerMap> {
  private handlers: Partial<Record<keyof T, Array<(...args: any[]) => void>>> = {};

  /** @param context 回调执行时的 this，通常是持有事件总线的实例 */
  constructor(private readonly context?: unknown) {}

  on<K extends keyof T>(type: K, callback: T[K]): void {
    (this.handlers[type] ||= []).push(callback);
  }

  off<K extends keyof T>(type: K): void {
    delete this.handlers[type];
  }

  trigger<K extends keyof T>(type: K, ...args: Parameters<T[K]>): void {
    const list = this.handlers[type];
    if (!list) return;
    for (const fn of [...list]) {
      try {
        fn.apply((this.context as object | undefined) ?? this, args);
      } catch (error) {
        console.warn('[ofdh5] 事件回调异常:', type, error);
      }
    }
  }

  clear(): void {
    this.handlers = {};
  }
}
```

```ts
import { unzipSync } from 'fflate';
import { OfdError } from '../errors';

export function isZip(buffer: Uint8Array): boolean {
  return buffer.length >= 4 && buffer[0] === 0x50 && buffer[1] === 0x4b && buffer[2] === 0x03 && buffer[3] === 0x04;
}

export function normalizePath(path: string): string {
  const parts: string[] = [];
  for (const segment of path.replace(/\\/g, '/').split('/')) {
    if (!segment || segment === '.') continue;
    if (segment === '..') parts.pop();
    else parts.push(segment);
  }
  return parts.join('/');
}

export function resolvePath(base: string, relative: string): string {
  if (relative.startsWith('/')) return normalizePath(relative);
  const dir = base.includes('/') ? base.slice(0, base.lastIndexOf('/')) : '';
  return normalizePath(dir ? `${dir}/${relative}` : relative);
}

export class OfdZip {
  private readonly files = new Map<string, Uint8Array>();

  constructor(buffer: Uint8Array) {
    if (!isZip(buffer)) {
      throw new OfdError('E_NOT_ZIP', '文件不是合法的 OFD 包（zip 魔数不匹配）');
    }
    let entries: Record<string, Uint8Array>;
    try {
      entries = unzipSync(buffer);
    } catch (error) {
      throw new OfdError('E_NOT_ZIP', `zip 解压失败：${(error as Error).message}`);
    }
    for (const [name, data] of Object.entries(entries)) {
      this.files.set(normalizePath(name), data);
    }
  }

  has(path: string): boolean {
    return this.files.has(normalizePath(path));
  }

  read(path: string): Uint8Array | null {
    return this.files.get(normalizePath(path)) ?? null;
  }

  readText(path: string): string {
    const data = this.read(path);
    if (!data) throw new OfdError('E_NO_ENTRY', `缺少文件：${path}`);
    return new TextDecoder('utf-8').decode(data);
  }

  list(): string[] {
    return [...this.files.keys()];
  }
}
```

```ts
import { OfdError } from '../errors';

export function parseXml(source: string, label: string): Document {
  const doc = new DOMParser().parseFromString(source, 'application/xml');
  const errors = doc.getElementsByTagName('parsererror');
  if (errors.length > 0) {
    const detail = (errors[0].textContent ?? '').replace(/\s+/g, ' ').slice(0, 160);
    throw new OfdError('E_PARSE_XML', `${label} 解析失败：${detail}`);
  }
  return doc;
}

export function firstByLocalName(root: Element | Document, localName: string): Element | null {
  const list = root.getElementsByTagName('*');
  for (let i = 0; i < list.length; i++) {
    if (list[i].localName === localName) return list[i];
  }
  return null;
}

export function allByLocalName(root: Element | Document, localName: string): Element[] {
  const out: Element[] = [];
  const list = root.getElementsByTagName('*');
  for (let i = 0; i < list.length; i++) {
    if (list[i].localName === localName) out.push(list[i]);
  }
  return out;
}

export function directChild(element: Element, localName: string): Element | null {
  const children = element.children;
  for (let i = 0; i < children.length; i++) {
    if (children[i].localName === localName) return children[i];
  }
  return null;
}

export function attr(element: Element | null, name: string): string | null {
  if (!element) return null;
  if (element.hasAttribute(name)) return element.getAttribute(name);
  const attributes = element.attributes;
  for (let i = 0; i < attributes.length; i++) {
    if (attributes[i].localName === name) return attributes[i].value;
  }
  return null;
}

export function num(value: string | null, fallback = 0): number {
  const parsed = Number.parseFloat(value ?? '');
  return Number.isFinite(parsed) ? parsed : fallback;
}

export function nums(value: string | null): number[] {
  if (!value) return [];
  return value
    .trim()
    .split(/\s+/)
    .map((item) => Number.parseFloat(item))
    .filter((item) => Number.isFinite(item));
}
```

```ts
import { OfdError } from '../errors';
import { resolvePath } from '../zip';
import { allByLocalName, attr, firstByLocalName, nums, parseXml } from './xml';

export interface PageRef {
  id: string;
  loc: string;
}

export interface DocumentMeta {
  pages: PageRef[];
  /** mm */
  width: number;
  /** mm */
  height: number;
}

export function parseDocumentMeta(source: string, docRoot: string): DocumentMeta {
  const doc = parseXml(source, 'Document.xml');

  const box = nums(firstByLocalName(doc, 'PhysicalBox')?.textContent ?? '');
  const width = box[2] ?? 210;
  const height = box[3] ?? 297;

  const pageEls = allByLocalName(doc, 'Page');
  if (pageEls.length === 0) {
    throw new OfdError('E_NO_ENTRY', 'Document.xml 中缺少 Pages/Page');
  }

  const pages: PageRef[] = pageEls.map((el, index) => {
    const baseLoc = attr(el, 'BaseLoc');
    if (!baseLoc) {
      throw new OfdError('E_PARSE_XML', `第 ${index + 1} 页缺少 BaseLoc`);
    }
    return {
      id: attr(el, 'ID') ?? String(index + 1),
      loc: resolvePath(docRoot, baseLoc),
    };
  });

  return { pages, width, height };
}
```

```ts
import type { RgbColor } from '../types';

function component(value: number, max: number): number {
  const scaled = value <= 1 ? value * max : value;
  return Math.max(0, Math.min(max, Math.round(scaled)));
}

export function parseColor(value: string | null, alpha?: string | null): RgbColor | undefined {
  if (!value || !value.trim()) return undefined;
  const parts = value
    .trim()
    .split(/\s+/)
    .map((item) => Number.parseFloat(item))
    .filter((item) => Number.isFinite(item));
  if (parts.length === 0) return undefined;

  let r = 0;
  let g = 0;
  let b = 0;

  if (parts.length === 1) {
    r = g = b = component(parts[0], 255);
  } else if (parts.length === 3) {
    r = component(parts[0], 255);
    g = component(parts[1], 255);
    b = component(parts[2], 255);
  } else if (parts.length >= 4) {
    const [c, m, y, k] = parts.map((item) => (item > 1 ? item / 100 : item));
    r = Math.round(255 * (1 - c) * (1 - k));
    g = Math.round(255 * (1 - m) * (1 - k));
    b = Math.round(255 * (1 - y) * (1 - k));
  }

  let a = 1;
  if (alpha != null && alpha !== '') {
    const parsed = Number.parseFloat(alpha);
    if (Number.isFinite(parsed)) a = parsed > 1 ? Math.min(1, parsed / 255) : Math.max(0, parsed);
  }

  return { r, g, b, a };
}

export function toCssColor(color: RgbColor): string {
  return `rgba(${color.r},${color.g},${color.b},${color.a})`;
}
```

```ts
const ARG_COUNT: Record<string, number> = { M: 2, L: 2, B: 6, Q: 4, A: 7, C: 0, S: 2 };
const SVG_OP: Record<string, string> = { M: 'M', L: 'L', B: 'C', Q: 'Q', A: 'A', C: 'Z', S: 'M' };

/**
 * OFD AbbreviatedData → SVG path 字符串（可直接 new Path2D(d)）
 * M/L/C 与 SVG 同名，B(三次贝塞尔)→C，Q(二次)→Q，A(弧)→A，C(闭合)→Z，S 按 M 处理。
 */
export function abbreviatedToSvgPath(source: string): string {
  const tokens = source.trim().split(/\s+/).filter(Boolean);
  const out: string[] = [];
  let i = 0;
  while (i < tokens.length) {
    const op = tokens[i];
    const svgOp = SVG_OP[op];
    if (!svgOp) {
      i += 1;
      continue;
    }
    const argc = ARG_COUNT[op];
    out.push(svgOp);
    for (let k = 0; k < argc; k++) {
      const token = tokens[i + 1 + k];
      if (token === undefined || SVG_OP[token]) break;
      out.push(token);
    }
    i += 1 + argc;
  }
  return out.join(' ');
}
```

```ts
import type { Glyph, Page, PageItem, PageResources, TextItem } from '../types';
import { parseColor } from './color';
import { abbreviatedToSvgPath } from './path-data';
import { allByLocalName, attr, directChild, firstByLocalName, num, nums, parseXml } from './xml';

export interface PageSize {
  width: number;
  height: number;
}

function parseBoundary(value: string | null): [number, number, number, number] {
  const parts = nums(value);
  return [parts[0] ?? 0, parts[1] ?? 0, parts[2] ?? 0, parts[3] ?? 0];
}

function parseCtm(value: string | null): number[] | undefined {
  const parts = nums(value);
  if (parts.length < 6) return undefined;
  return parts.slice(0, 6);
}

function parseText(el: Element, pageWidth: number): TextItem {
  const textCode = directChild(el, 'TextCode');
  const glyphEls = textCode ? allByLocalName(textCode, 'Glyph') : [];

  let glyphs: Glyph[] = [];
  let text = '';

  if (glyphEls.length > 0) {
    glyphs = glyphEls.map((g) => ({
      x: num(attr(g, 'X')),
      y: num(attr(g, 'Y')),
      code: num(attr(g, 'Code')),
      deltaX: attr(g, 'DeltaX') != null ? num(attr(g, 'DeltaX')) : undefined,
    }));
    text = glyphs.map((g) => String.fromCodePoint(g.code)).join('');
  } else if (textCode) {
    const chars = Array.from(textCode.textContent ?? '');
    const deltaXs = nums(attr(textCode, 'DeltaX'));
    const deltaYs = nums(attr(textCode, 'DeltaY'));
    let x = num(attr(textCode, 'X'));
    const baseY = num(attr(textCode, 'Y'));
    glyphs = chars.map((char, index) => {
      const dx = deltaXs.length > 1 ? deltaXs[index] : deltaXs[0];
      const dy = deltaYs.length > 1 ? deltaYs[index] : deltaYs[0];
      const glyph: Glyph = {
        x,
        y: baseY + (dy ?? 0),
        code: char.codePointAt(0) ?? 0,
        deltaX: dx,
        deltaY: dy,
      };
      x += dx ?? 0;
      return glyph;
    });
    text = chars.join('');
  }

  const fillEl = directChild(el, 'FillColor');
  const strokeEl = directChild(el, 'StrokeColor');
  const hScale = attr(el, 'HScale');

  return {
    kind: 'text',
    boundary: parseBoundary(attr(el, 'Boundary')),
    ctm: parseCtm(attr(el, 'CTM')),
    fontRef: attr(el, 'Font') ?? '',
    size: num(attr(el, 'Size'), 3.5),
    fill: parseColor(attr(fillEl, 'Value'), attr(fillEl, 'Alpha')),
    stroke: parseColor(attr(strokeEl, 'Value'), attr(strokeEl, 'Alpha')),
    hScale: hScale == null ? undefined : num(hScale, 100),
    weight: attr(el, 'Weight') == null ? undefined : num(attr(el, 'Weight'), 400),
    italic: attr(el, 'Italic') === 'true',
    glyphs,
    text,
  } as TextItem & { stroke?: unknown };
}

export function parsePage(
  source: string,
  index: number,
  _resources: PageResources,
  fallback?: PageSize,
): Page {
  const doc = parseXml(source, `Content.xml(第 ${index + 1} 页)`);
  const root = doc.documentElement;

  const box = nums(firstByLocalName(root, 'PhysicalBox')?.textContent ?? '');
  const width = box[2] ?? fallback?.width ?? 210;
  const height = box[3] ?? fallback?.height ?? 297;

  const items: PageItem[] = [];
  for (const layer of allByLocalName(root, 'Layer')) {
    for (let i = 0; i < layer.children.length; i++) {
      const child = layer.children[i];
      if (child.localName === 'TextObject') items.push(parseText(child, width));
    }
  }

  return { index, width, height, items };
}

export { abbreviatedToSvgPath };
```

```ts
function mimeOf(file: string): string {
  const lower = file.toLowerCase();
  if (lower.endsWith('.png')) return 'image/png';
  if (lower.endsWith('.jpg') || lower.endsWith('.jpeg')) return 'image/jpeg';
  if (lower.endsWith('.gif')) return 'image/gif';
  if (lower.endsWith('.bmp')) return 'image/bmp';
  return 'application/octet-stream';
}

export function parseResourcesXml(source: string, baseLoc: string, resources: PageResources): PageResources {
  const doc = parseXml(source, 'Resources');
  const root = doc.documentElement;

  for (const font of allByLocalName(root, 'Font')) {
    const id = attr(font, 'ID');
    if (!id) continue;
    const fileEl = firstByLocalName(font, 'FontFile');
    const file = fileEl?.textContent?.trim();
    resources.fonts.set(id, {
      id,
      name: attr(font, 'FontName') ?? '',
      family: attr(font, 'FamilyName') ?? attr(font, 'FontName') ?? '',
      file: file ? resolvePath(baseLoc, file) : undefined,
    });
  }

  for (const media of allByLocalName(root, 'MultiMedia')) {
    const id = attr(media, 'ID');
    const file = firstByLocalName(media, 'MediaFile')?.textContent?.trim();
    if (!id || !file) continue;
    const resolved = resolvePath(baseLoc, file);
    resources.images.set(id, { id, file: resolved, mime: mimeOf(resolved) });
  }

  return resources;
}
```

```ts
export interface FontRegistry {
  register(id: string, data: Uint8Array, familyHint?: string): Promise<boolean>;
  family(id: string, familyHint?: string): string;
  has(id: string): boolean;
}

export function createFontRegistry(fallback = 'sans-serif'): FontRegistry {
  const families = new Map<string, string>();

  return {
    async register(id, data, familyHint) {
      if (families.has(id)) return true;
      const failFamily = familyHint ? `"${familyHint}", ${fallback}` : fallback;

      if (typeof FontFace === 'undefined' || typeof document === 'undefined' || !document.fonts) {
        families.set(id, failFamily);
        return false;
      }

      try {
        const family = `ofdh5-${id}`;
        const source = data.slice().buffer;
        const face = new FontFace(family, source);
        await face.load();
        document.fonts.add(face);
        families.set(id, `"${family}"`);
        return true;
      } catch {
        families.set(id, failFamily);
        return false;
      }
    },

    family(id, familyHint) {
      return families.get(id) ?? (familyHint ? `"${familyHint}", ${fallback}` : fallback);
    },

    has(id) {
      return families.has(id);
    },
  };
}
```

```ts
export interface PageSizeLike {
  width: number;
  height: number;
}

export interface Viewport {
  cssWidth: number;
  cssHeight: number;
  pxPerMm: number;
  dpr: number;
}

export function createViewport(page: PageSizeLike, containerWidth: number, scale: number, dpr: number): Viewport {
  const cssWidth = Math.max(1, Math.round(containerWidth * scale));
  const pxPerMm = cssWidth / page.width;
  return {
    cssWidth,
    cssHeight: Math.round(page.height * pxPerMm),
    pxPerMm,
    dpr,
  };
}

export function clampScale(scale: number, min: number, max: number): number {
  if (!Number.isFinite(scale)) return min;
  return Math.min(max, Math.max(min, scale));
}
```

```ts
export class LruCache<K, V> {
  private readonly map = new Map<K, V>();

  constructor(private readonly limit: number, private readonly onEvict?: (value: V, key: K) => void) {}

  get(key: K): V | undefined {
    if (!this.map.has(key)) return undefined;
    const value = this.map.get(key) as V;
    this.map.delete(key);
    this.map.set(key, value);
    return value;
  }

  set(key: K, value: V): void {
    if (this.map.has(key)) this.map.delete(key);
    this.map.set(key, value);
    while (this.map.size > this.limit) {
      const oldest = this.map.keys().next();
      if (oldest.done) break;
      const keyToEvict = oldest.value;
      const valueToEvict = this.map.get(keyToEvict) as V;
      this.map.delete(keyToEvict);
      this.onEvict?.(valueToEvict, keyToEvict);
    }
  }

  has(key: K): boolean {
    return this.map.has(key);
  }

  delete(key: K): void {
    if (!this.map.has(key)) return;
    const value = this.map.get(key) as V;
    this.map.delete(key);
    this.onEvict?.(value, key);
  }

  clear(): void {
    for (const key of [...this.map.keys()]) this.delete(key);
  }

  keys(): K[] {
    return [...this.map.keys()];
  }

  get size(): number {
    return this.map.size;
  }
}
```

```ts
  return {
    async load(id, path) {
      const hit = cache.get(id);
      if (hit) {
        cache.delete(id);
        cache.set(id, hit);
        return hit;
      }
      const running = pending.get(id);
      if (running) return running;

      const data = source.read(path);
      if (!data) return null;

      const task = decode(toBlob(data, ''))
        .then((image) => {
          if (image) {
            cache.set(id, image);
            evictIfNeeded();
          }
          return image;
        })
        .finally(() => pending.delete(id));

      pending.set(id, task);
      return task;
    },
    get(id) {
      return cache.get(id);
    },
    clear() {
      for (const image of cache.values()) {
        if (image && 'close' in image) image.close();
      }
      cache.clear();
    },
    get size() {
      return cache.size;
    },
  };
}
```

```ts
import type { ImageItem, Page, PageItem, PathItem, RgbColor, TextItem } from '../types';
import type { Viewport } from './viewport';
import { toCssColor } from '../parser/color';

export interface PaintOptions {
  background: string;
  fontFamily: (fontRef: string) => string;
  getImage: (item: ImageItem) => CanvasImageSource | null;
  onItemError?: (item: PageItem, error: unknown) => void;
}

function applyCtm(ctx: CanvasRenderingContext2D, item: PageItem): void {
  const [x, y] = item.boundary;
  ctx.translate(x, y);
  if (item.ctm && item.ctm.length >= 6) {
    const [a, b, c, d, e, f] = item.ctm;
    ctx.transform(a, b, c, d, e, f);
  }
}

function fillColor(color: RgbColor | undefined): string | null {
  return color ? toCssColor(color) : null;
}

function drawText(ctx: CanvasRenderingContext2D, item: TextItem, opts: PaintOptions): void {
  ctx.save();
  applyCtm(ctx, item);
  if (item.hScale && item.hScale !== 100) ctx.scale(item.hScale / 100, 1);

  const family = opts.fontFamily(item.fontRef);
  const weight = item.weight ? `${item.weight} ` : '';
  const style = item.italic ? 'italic ' : '';
  ctx.font = `${style}${weight}${item.size}px ${family}`;
  ctx.textBaseline = 'alphabetic';

  const color = fillColor(item.fill) ?? 'rgba(0,0,0,1)';
  ctx.fillStyle = color;
  if (item.stroke) {
    ctx.strokeStyle = fillColor(item.stroke) ?? color;
    ctx.lineWidth = item.size * 0.02;
  }

  let cursor = 0;
  for (let i = 0; i < item.glyphs.length; i++) {
    const glyph = item.glyphs[i];
    if (glyph.deltaX != null || i === 0) cursor = glyph.x;
    const char = item.text[i];
    if (char) {
      ctx.fillText(char, cursor, glyph.y);
      if (item.stroke) ctx.strokeText(char, cursor, glyph.y);
    }
    cursor += glyph.deltaX ?? (char ? ctx.measureText(char).width : 0);
  }
  ctx.restore();
}

function drawPath(ctx: CanvasRenderingContext2D, item: PathItem): void {
  if (typeof Path2D === 'undefined') return;
  const path = new Path2D(item.d);
  ctx.save();
  applyCtm(ctx, item);
  if (item.fill) {
    ctx.fillStyle = toCssColor(item.fill);
    ctx.fill(path, item.rule);
  }
  if (item.stroke) {
    ctx.strokeStyle = toCssColor(item.stroke);
    ctx.lineWidth = item.lineWidth ?? 1;
    ctx.stroke(path);
  }
  ctx.restore();
}

function drawImage(ctx: CanvasRenderingContext2D, item: ImageItem, opts: PaintOptions): void {
  const image = opts.getImage(item);
  if (!image) return;
  const [x, y, w, h] = item.boundary;
  ctx.save();
  applyCtm(ctx, item);
  ctx.drawImage(image, 0, 0, w, h);
  ctx.restore();
}

export function paintPage(
  ctx: CanvasRenderingContext2D,
  page: Page,
  vp: Viewport,
  opts: PaintOptions,
): void {
  const scale = vp.dpr * vp.pxPerMm;
  ctx.setTransform(scale, 0, 0, scale, 0, 0);
  ctx.fillStyle = opts.background;
  ctx.fillRect(0, 0, page.width, page.height);

  for (const item of page.items) {
    try {
      if (item.kind === 'text') drawText(ctx, item, opts);
      else if (item.kind === 'path') drawPath(ctx, item);
      else drawImage(ctx, item, opts);
    } catch (error) {
      opts.onItemError?.(item, error);
      console.warn('[ofdh5] 对象绘制失败:', item.kind, error);
    }
  }
}
```

export interface Point {
  x: number;
  y: number;
}

export interface PinchZoomOptions {
  minScale: number;
  maxScale: number;
  tapZoomFactor: number;
  enabled: boolean;
  onZoom?: (scale: number, origin: Point) => void;
  onZoomStart?: () => void;
  onZoomEnd?: (scale: number) => void;
}

export function distance(a: Point, b: Point): number {
  return Math.hypot(a.x - b.x, a.y - b.y);
}

export function center(a: Point, b: Point): Point {
  return { x: (a.x + b.x) / 2, y: (a.y + b.y) / 2 };
}

export function clamp(value: number, min: number, max: number): number {
  return Math.min(max, Math.max(min, value));
}

export function pinchScale(startDistance: number, currentDistance: number, baseScale: number): number {
  if (startDistance <= 0) return baseScale;
  return (currentDistance / startDistance) * baseScale;
}

export class PinchZoom {
  private scaleValue = 1;
  private baseWidth = 0;
  private baseHeight = 0;
  private startDistance = 0;
  private startScale = 1;
  private lastTapAt = 0;

  constructor(
    private readonly host: HTMLElement,
    private readonly target: HTMLElement,
    private readonly sizer: HTMLElement,
    private readonly options: PinchZoomOptions,
  ) {
    this.host.addEventListener('touchstart', this.handleStart, { passive: true });
    this.host.addEventListener('touchmove', this.handleMove, { passive: false });
    this.host.addEventListener('touchend', this.handleEnd);
  }

  get scale(): number {
    return this.scaleValue;
  }

  setBaseSize(width: number, height: number): void {
    this.baseWidth = width;
    this.baseHeight = height;
    this.render(this.scaleValue);
  }

  private handleStart = (event: TouchEvent): void => {
    if (!this.options.enabled) return;
    if (event.touches.length === 2) {
      this.startDistance = distance(this.pointOf(event.touches[0]), this.pointOf(event.touches[1]));
      this.startScale = this.scaleValue;
      this.options.onZoomStart?.();
      return;
    }
    if (event.touches.length === 1) {
      const now = Date.now();
      if (now - this.lastTapAt < 300) {
        this.lastTapAt = 0;
        const touch = event.touches[0];
        const next = this.scaleValue > this.options.minScale + 0.01
          ? this.options.minScale
          : clamp(this.options.tapZoomFactor, this.options.minScale, this.options.maxScale);
        this.zoomAt(next, touch.clientX, touch.clientY);
      } else {
        this.lastTapAt = now;
      }
    }
  };

  private handleMove = (event: TouchEvent): void => {
    if (!this.options.enabled || event.touches.length !== 2 || this.startDistance === 0) return;
    event.preventDefault();
    const currentDistance = distance(this.pointOf(event.touches[0]), this.pointOf(event.touches[1]));
    const next = clamp(
      pinchScale(this.startDistance, currentDistance, this.startScale),
      this.options.minScale,
      this.options.maxScale,
    );
    const mid = center(this.pointOf(event.touches[0]), this.pointOf(event.touches[1]));
    this.zoomAt(next, mid.x, mid.y);
  };

  private handleEnd = (): void => {
    if (this.startDistance === 0) return;
    this.startDistance = 0;
    this.options.onZoomEnd?.(this.scaleValue);
  };

  private pointOf(touch: Touch): Point {
    return { x: touch.clientX, y: touch.clientY };
  }

  /**
   * 以 (clientX, clientY) 为锚点缩放：按比例换算内容坐标后回填滚动位置，
   * 保证手指下方的点保持不动。
   */
  zoomAt(nextScale: number, clientX: number, clientY: number): void {
    const clamped = clamp(nextScale, this.options.minScale, this.options.maxScale);
    const ratio = clamped / this.scaleValue;
    if (Math.abs(ratio - 1) < 0.0005) return;

    const rect = this.host.getBoundingClientRect();
    const offsetX = clientX - rect.left;
    const offsetY = clientY - rect.top;
    const contentX = this.host.scrollLeft + offsetX;
    const contentY = this.host.scrollTop + offsetY;

    this.render(clamped);
    this.host.scrollLeft = contentX * ratio - offsetX;
    this.host.scrollTop = contentY * ratio - offsetY;
    this.options.onZoom?.(clamped, { x: contentX * ratio, y: contentY * ratio });
  }

  setScale(nextScale: number): void {
    const rect = this.host.getBoundingClientRect();
    this.zoomAt(nextScale, rect.left + rect.width / 2, rect.top + rect.height / 2);
  }

  /** 只做 CSS 变换与滚动容器尺寸同步，不改滚动位置 */
  private render(scale: number): void {
    this.scaleValue = scale;
    this.target.style.transformOrigin = '0 0';
    this.target.style.transform = `scale(${scale})`;
    this.sizer.style.width = `${Math.round(this.baseWidth * scale)}px`;
    this.sizer.style.height = `${Math.round(this.baseHeight * scale)}px`;
  }

  reset(): void {
    this.render(1);
    this.host.scrollTop = 0;
    this.host.scrollLeft = 0;
    this.options.onZoom?.(1, { x: 0, y: 0 });
  }

  destroy(): void {
    this.host.removeEventListener('touchstart', this.handleStart);
    this.host.removeEventListener('touchmove', this.handleMove);
    this.host.removeEventListener('touchend', this.handleEnd);
  }
}
> 说明：用「缩放 target + 同步 sizer 尺寸 + 回填滚动位置」实现锚点缩放，不依赖 transform-origin 与布局的交互，iOS Safari 上滚动范围也正确。缩放不做缓动，双击/双指即时生效——M4 验收若觉得生硬再加 CSS transition。

```ts
export function pageAtOffset(
  offsets: number[],
  heights: number[],
  scrollTop: number,
  viewportHeight: number,
): number {
  if (offsets.length === 0) return 0;
  const probe = scrollTop + viewportHeight / 3;
  let result = 0;
  for (let i = 0; i < offsets.length; i++) {
    if (probe >= offsets[i]) result = i;
    else break;
  }
  const last = offsets.length - 1;
  return Math.max(0, Math.min(last, result));
}

const schedule: (cb: () => void) => number = typeof requestAnimationFrame === 'function'
  ? (cb) => requestAnimationFrame(cb)
  : (cb) => setTimeout(cb, 16) as unknown as number;

const unschedule: (id: number) => void = typeof cancelAnimationFrame === 'function'
  ? (id) => cancelAnimationFrame(id)
  : (id) => clearTimeout(id);

export class ScrollWatcher {
  private rafId = 0;
  private readonly handler = () => {
    if (this.rafId) return;
    this.rafId = schedule(() => {
      this.rafId = 0;
      this.onScroll(this.host.scrollTop);
    });
  };

  constructor(private readonly host: HTMLElement, private readonly onScroll: (scrollTop: number) => void) {
    this.host.addEventListener('scroll', this.handler, { passive: true });
  }

  destroy(): void {
    unschedule(this.rafId);
    this.host.removeEventListener('scroll', this.handler);
  }
}
```

```ts
interface ResolvedOptions {

export class Ofdh5 {
  readonly version = '0.1.0';
  totalNum = 0;
  currentNum = 1;

  private readonly bus: EventBus<Ofdh5Events>;
  private readonly opts: ResolvedOptions;
  private readonly container: HTMLElement;

  private chrome: Chrome | null = null;
  private zoom: PinchZoom | null = null;
  private watcher: ScrollWatcher | null = null;

  private zip: OfdZip | null = null;
  private buffer: Uint8Array | null = null;
  private meta: DocumentMeta | null = null;
  private resources: PageResources = createResources();
  private fonts: FontRegistry;
  private images: ImageCache | null = null;

  private pageCache = new Map<number, Page>();
  private placeholders: HTMLElement[] = [];
  private rendered = new Set<number>();
  private rendering = new Set<number>();
  private skippedObjects = 0;

  private status: OfdStatus['status'] = 'init';
  private lastError: OfdStatus['lastError'] = null;
  private startedAt = 0;
  private resizeHandler: (() => void) | null = null;

  private resolveLoaded: (() => void) | null = null;
  /** 测试与外部等待加载结束用 */
  readonly loaded: Promise<void>;

  constructor(container: HTMLElement, options?: Ofdh5Options) {
    this.container = container;
    this.opts = resolveOptions(options);
    this.bus = new EventBus<Ofdh5Events>(this);
    this.fonts = createFontRegistry(this.opts.fontFallback);
    this.loaded = new Promise<void>((resolve) => {
      this.resolveLoaded = resolve;
    });
    void this.init();
  }

  private async init(): Promise<void> {
    this.startedAt = performance.now();
    this.bus.trigger('init');
    try {
      if (!this.opts.url && !this.opts.data) {
        throw new OfdError('E_NETWORK', '缺少 url 或 data');
      }
      this.chrome = createChrome(this.container, {
        loadingBar: this.opts.loadingBar,
        pageNum: this.opts.pageNum,
        backTop: this.opts.backTop,
      });
      this.chrome.onBackTop(() => this.scrollToTop());
      await this.load();
    } catch (error) {
      this.fail(error);
    } finally {
      this.complete();
    }
  }

  private async load(): Promise<void> {
    const buffer = await this.readSource();
    this.buffer = buffer;

    const zip = new OfdZip(buffer);
    if (zip.has('Encrypt.xml')) {
      throw new OfdError('E_ENCRYPTED', '暂不支持加密 OFD');
    }
    this.zip = zip;

    const pkg = parseOfdPackage(zip.readText('OFD.xml'));
    const docRoot = pkg.docRoots[0];
    this.meta = parseDocumentMeta(zip.readText(docRoot), docRoot);
    this.totalNum = this.meta.pages.length;
    this.status = 'ready';

    this.images = createImageCache({ read: (path) => zip.read(path) });

    this.buildPlaceholders();
    this.attachInteractions();
    this.chrome!.setPage(1, this.totalNum);
    this.bus.trigger('ready');

    await this.renderVisible();
    this.chrome!.hideLoading();
    this.bus.trigger('success', 'load success', performance.now() - this.startedAt);

    if (this.opts.goto > 0) this.goToPage(this.opts.goto);
  }

  private async readSource(): Promise<Uint8Array> {
    if (this.opts.data) {
      if (this.opts.data instanceof ArrayBuffer) return new Uint8Array(this.opts.data);
      if (this.opts.data instanceof Uint8Array) return this.opts.data;
      return new Uint8Array(await (this.opts.data as Blob).arrayBuffer());
    }
    const controller = typeof AbortController === 'function' ? new AbortController() : null;
    const timer = setTimeout(() => controller?.abort(), this.opts.timeout);
    try {
      const response = await fetch(this.opts.url as string, controller ? { signal: controller.signal } : {});
      if (!response.ok) throw new OfdError('E_NETWORK', `下载失败：HTTP ${response.status}`);
      return new Uint8Array(await response.arrayBuffer());
    } catch (error) {
      if (error instanceof OfdError) throw error;
      throw new OfdError('E_NETWORK', `下载失败：${(error as Error).message}`);
    } finally {
      clearTimeout(timer);
    }
  }

  private buildPlaceholders(): void {
    const viewer = this.chrome!.viewer;
    viewer.innerHTML = '';
    this.placeholders = [];
    const width = this.contentWidth();
    const ratio = (this.meta?.height ?? 297) / (this.meta?.width ?? 210);

    for (let i = 0; i < this.totalNum; i++) {
      const page = document.createElement('div');
      page.className = 'ofdh5-page';
      page.dataset.index = String(i);
      page.style.height = `${Math.round(width * ratio)}px`;
      page.style.marginBottom = `${this.opts.pageGap}px`;
      viewer.appendChild(page);
      this.placeholders.push(page);
    }
  }

  private attachInteractions(): void {
    this.zoom = new PinchZoom(this.chrome!.scroller, this.chrome!.viewer, this.chrome!.sizer, {
      minScale: this.opts.minZoom,
      maxScale: this.opts.maxZoom,
      tapZoomFactor: this.opts.tapZoomFactor,
      enabled: this.opts.zoomEnable,
      onZoom: (scale) => this.bus.trigger('zoom', scale),
      onZoomStart: () => this.bus.trigger('zoomStart'),
      onZoomEnd: (scale) => this.bus.trigger('zoomEnd', scale),
    });
    this.zoom.setBaseSize(this.chrome!.viewer.offsetWidth, this.chrome!.viewer.offsetHeight);

    this.watcher = new ScrollWatcher(this.chrome!.scroller, (scrollTop) => {
      const current = this.pageIndexAt(scrollTop);
      if (current !== this.currentNum - 1) {
        this.currentNum = current + 1;
        this.chrome!.setPage(this.currentNum, this.totalNum);
      }
      this.bus.trigger('scroll', { scrollTop, currentNum: this.currentNum });
      void this.renderVisible();
    });

    if (this.opts.resize && typeof window !== 'undefined') {
      this.resizeHandler = () => {
        this.rendered.clear();
        this.buildPlaceholders();
        void this.renderVisible();
      };
      window.addEventListener('resize', this.resizeHandler);
    }
  }

  /** 注册当前已收集到但尚未注册的内嵌字体；每解析一页都调用一次 */
  private async ensureFonts(pageNum: number): Promise<void> {
    if (!this.zip) return;
    for (const [id, font] of this.resources.fonts) {
      if (this.fonts.has(id) || !font.file) continue;
      const data = this.zip.read(font.file);
      if (!data) continue;
      const ok = await this.fonts.register(id, data, font.family || font.name);
      if (!ok) this.bus.trigger('fontFallback', pageNum, font.name || font.family);
    }
  }

  private pageIndexAt(scrollTop: number): number {
    const offsets = this.placeholders.map((el) => el.offsetTop);
    const heights = this.placeholders.map((el) => el.offsetHeight);
    return pageAtOffset(offsets, heights, scrollTop, this.chrome!.scroller.clientHeight || 600);
  }

  private contentWidth(): number {
    const width = this.chrome!.scroller.clientWidth || this.container.clientWidth || 375;
    return width - 16;
  }

  private dpr(): number {
    const raw = typeof window !== 'undefined' ? window.devicePixelRatio || 1 : 1;
    return Math.min(raw, this.opts.maxDpr);
  }

  private async getPage(index: number): Promise<Page> {
    const cached = this.pageCache.get(index);
    if (cached) return cached;
    if (!this.zip || !this.meta) throw new OfdError('E_NO_ENTRY', '包未加载');

    const loc = this.meta.pages[index].loc;
    const parsed = parsePage(this.zip.readText(loc), index, this.resources, {
      width: this.meta.width,
      height: this.meta.height,
    }, loc);

    this.pageCache.set(index, parsed);
    return parsed;
  }

  private async prefetchImages(page: Page): Promise<void> {
    if (!this.images) return;
    const jobs: Array<Promise<unknown>> = [];
    for (const item of page.items) {
      if (item.kind !== 'image') continue;
      const resource = this.resources.images.get(item.resId);
      if (!resource) continue;
      jobs.push(this.images.load(resource.id, resource.file));
    }
    await Promise.all(jobs);
  }

  private async renderPage(index: number): Promise<void> {
    if (index < 0 || index >= this.totalNum) return;
    if (this.rendered.has(index) || this.rendering.has(index)) return;
    this.rendering.add(index);
    const pageStarted = performance.now();
    const placeholder = this.placeholders[index];

    try {
      const page = await this.getPage(index);
      await this.ensureFonts(index + 1);
      const width = this.contentWidth() * this.opts.scale;
      const viewport = createViewport(page, width, 1, this.dpr());
      await this.prefetchImages(page);

      const canvas = document.createElement('canvas');
      canvas.width = Math.round(viewport.cssWidth * viewport.dpr);
      canvas.height = Math.round(viewport.cssHeight * viewport.dpr);
      canvas.style.width = `${viewport.cssWidth}px`;
      canvas.style.height = `${viewport.cssHeight}px`;

      const ctx = canvas.getContext('2d');
      if (!ctx) throw new OfdError('E_CANVAS', '无法创建 canvas 上下文');

      paintPage(ctx, page, viewport, {
        background: this.opts.backgroundColor,
        fontFamily: (fontRef) => this.fonts.family(fontRef, this.resources.fonts.get(fontRef)?.family),
        getImage: (item) => {
          const decoded = this.images?.get(item.resId);
          return (decoded as DecodedImage | undefined) ?? null;
        },
        onItemError: () => {
          this.skippedObjects += 1;
        },
      });

      placeholder.innerHTML = '';
      placeholder.appendChild(canvas);
      this.rendered.add(index);
      this.bus.trigger('render', index + 1, performance.now() - pageStarted, performance.now() - this.startedAt);
    } catch (error) {
      const err = toOfdError(error, 'E_RENDER');
      placeholder.innerHTML = `<div class="ofdh5-page-error">第 ${index + 1} 页渲染失败</div>`;
      this.bus.trigger('pageError', index + 1, err);
    } finally {
      this.rendering.delete(index);
    }
  }

  private async renderVisible(): Promise<void> {
    const current = this.currentNum - 1;
    const targets = this.opts.lazyLoad
      ? [current - 1, current, current + 1]
      : Array.from({ length: this.totalNum }, (_, i) => i);
    for (const index of targets) {
      if (index >= 0 && index < this.totalNum) await this.renderPage(index);
    }
  }

  private fail(error: unknown): void {
    const err = toOfdError(error, 'E_RENDER');
    this.status = 'error';
    this.lastError = { code: err.code, message: err.message };
    if (this.chrome) {
      this.chrome.viewer.innerHTML = `<div class="ofdh5-page-error">加载失败（${err.code}）</div>`;
    }
    this.bus.trigger('error', `${err.code}: ${err.message}`, performance.now() - this.startedAt);
  }

  private complete(): void {
    if (this.status !== 'error') this.status = 'ready';
    this.bus.trigger(
      'complete',
      this.status === 'error' ? 'error' : 'success',
      this.lastError ? this.lastError.message : 'ok',
      performance.now() - this.startedAt,
    );
    this.resolveLoaded?.();
  }

  on<K extends keyof Ofdh5Events>(type: K, callback: Ofdh5Events[K]): void {
    this.bus.on(type, callback);
  }

  off<K extends keyof Ofdh5Events>(type: K): void {
    this.bus.off(type);
  }

  trigger<K extends keyof Ofdh5Events>(type: K, ...args: Parameters<Ofdh5Events[K]>): void {
    this.bus.trigger(type, ...args);
  }

  getPageCount(): number {
    return this.totalNum;
  }

  getPageSize(index: number): { width: number; height: number } {
    const page = this.pageCache.get(index - 1);
    if (page) return { width: page.width, height: page.height };
    return { width: this.meta?.width ?? 210, height: this.meta?.height ?? 297 };
  }

  getCacheStats(): { rendered: number; cachedPages: number } {
    return { rendered: this.rendered.size, cachedPages: this.rendered.size };
  }

  getStatus(): OfdStatus {
    return {
      status: this.status,
      pageCount: this.totalNum,
      currentPage: this.currentNum,
      scale: this.zoom?.scale ?? this.opts.scale,
      renderedPages: [...this.rendered].map((i) => i + 1).sort((a, b) => a - b),
      cachedPages: this.rendered.size,
      skippedObjects: this.skippedObjects,
      lastError: this.lastError,
    };
  }

  goToPage(pageNum: number): void {
    const index = Math.max(0, Math.min(this.totalNum - 1, Math.trunc(pageNum) - 1));
    const target = this.placeholders[index];
    if (!target || !this.chrome) return;
    this.chrome.scroller.scrollTop = target.offsetTop;
    this.currentNum = index + 1;
    this.chrome.setPage(this.currentNum, this.totalNum);
    void this.renderVisible();
  }

  nextPage(): void {
    this.goToPage(this.currentNum + 1);
  }

  prevPage(): void {
    this.goToPage(this.currentNum - 1);
  }

  scrollToTop(): void {
    if (!this.chrome) return;
    this.chrome.scroller.scrollTop = 0;
    this.currentNum = 1;
    this.chrome.setPage(1, this.totalNum);
    this.bus.trigger('backTop');
  }

  zoomIn(): void {
    this.zoom?.setScale((this.zoom?.scale ?? 1) + 0.1);
  }

  zoomOut(): void {
    this.zoom?.setScale((this.zoom?.scale ?? 1) - 0.1);
  }

  setScale(scale: number): void {
    this.opts.scale = Math.max(0.1, scale);
    this.rendered.clear();
    this.placeholders.forEach((el) => { el.innerHTML = ''; });
    void this.renderVisible();
  }

  setZoomEnabled(enabled: boolean): void {
    this.opts.zoomEnable = enabled;
    if (this.zoom) this.zoom.setEnabled(enabled);
    this.bus.trigger('zoomEnabled', enabled);
  }

  isZoomEnabled(): boolean {
    return this.opts.zoomEnable;
  }

  setScrollEnabled(enabled: boolean): void {
    this.opts.scrollEnable = enabled;
    if (this.chrome) this.chrome.scroller.style.overflowY = enabled ? 'auto' : 'hidden';
    this.bus.trigger('scrollEnabled', enabled);
  }

  isScrollEnabled(): boolean {
    return this.opts.scrollEnable;
  }

  setZoomConstraints(constraints: { minScale: number; maxScale: number; step?: number }): void {
    this.opts.minZoom = constraints.minScale;
    this.opts.maxZoom = constraints.maxScale;
  }

  getZoomConstraints(): { minScale: number; maxScale: number; step: number } {
    return { minScale: this.opts.minZoom, maxScale: this.opts.maxZoom, step: 0.1 };
  }

  isZooming(): boolean {
    return (this.zoom?.scale ?? 1) !== 1;
  }

  show(): void {
    this.container.style.display = '';
    this.bus.trigger('show');
  }

  hide(): void {
    this.container.style.display = 'none';
    this.bus.trigger('hide');
  }

  download(filename = 'document.ofd'): void {
    if (!this.buffer) return;
    const blob = new Blob([this.buffer.slice()], { type: 'application/ofd' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = filename;
    link.click();
    URL.revokeObjectURL(url);
  }

  destroy(): void {
    this.zoom?.destroy();
    this.watcher?.destroy();
    if (this.resizeHandler) window.removeEventListener('resize', this.resizeHandler);
    this.images?.clear();
    this.chrome?.destroy();
    this.pageCache.clear();
    this.rendered.clear();
    this.bus.trigger('destroy');
    this.bus.clear();
    this.status = 'destroyed';
    this.resolveLoaded?.();
  }
}
```

```ts
import type { Ofdh5Options } from '@ofdh5/core';

const OPTION_KEYS: Array<keyof Ofdh5Options> = [
  'url', 'pdfurl', 'data', 'goto', 'scale', 'scrollEnable', 'zoomEnable', 'resize',
  'lazyLoad', 'loadingBar', 'pageNum', 'backTop', 'tapZoomFactor', 'maxZoom', 'minZoom',
  'maxCachedPages', 'maxDpr', 'pageGap', 'backgroundColor', 'fontFallback', 'timeout',
];

export function toOptions(props: Partial<Ofdh5Options> & Record<string, unknown>): Ofdh5Options {
  const options: Ofdh5Options = {};
  for (const key of OPTION_KEYS) {
    const value = (props as Record<string, unknown>)[key];
    if (value !== undefined && value !== null) {
      (options as Record<string, unknown>)[key] = value;
    }
  }
  return options;
}
```

```vue
<template>
  <div ref="container" class="ofd-viewer"></div>

<script>
import { Ofdh5 } from '@ofdh5/core';
import { toOptions } from './options';

const EVENTS = ['init', 'ready', 'error', 'success', 'complete', 'render', 'zoom', 'zoomStart', 'zoomEnd', 'scroll', 'backTop'];

export default {
  name: 'OfdViewer',
  props: {
    url: { type: String, default: undefined },
    data: { type: [ArrayBuffer, Uint8Array, Blob], default: undefined },
    goto: { type: Number, default: undefined },
    scale: { type: Number, default: undefined },
    zoomEnable: { type: Boolean, default: undefined },
    scrollEnable: { type: Boolean, default: undefined },
    lazyLoad: { type: Boolean, default: undefined },
    loadingBar: { type: Boolean, default: undefined },
    pageNum: { type: Boolean, default: undefined },
    backTop: { type: Boolean, default: undefined },
    maxZoom: { type: Number, default: undefined },
    minZoom: { type: Number, default: undefined },
  },
  data() {
    return { instance: null, lastPage: 1 };
  },
  mounted() {
    this.create();
  },
  beforeDestroy() {
    this.destroy();
  },
  watch: {
    url() {
      this.create();
    },
    data() {
      this.create();
    },
  },
  methods: {
    create() {
      this.destroy();
      this.instance = new Ofdh5(this.$refs.container, toOptions(this.$props));
      this.instance.on('ready', (...args) => this.$emit('ready', ...args));
      this.instance.on('error', (...args) => this.$emit('error', ...args));
      this.instance.on('complete', (...args) => this.$emit('complete', ...args));
      this.instance.on('render', (...args) => this.$emit('render', ...args));
      this.instance.on('zoom', (...args) => this.$emit('zoom', ...args));
      this.instance.on('zoomStart', (...args) => this.$emit('zoom-start', ...args));
      this.instance.on('zoomEnd', (...args) => this.$emit('zoom-end', ...args));
      this.instance.on('scroll', (payload) => {
        this.$emit('scroll', payload);
        if (payload.currentNum !== this.lastPage) {
          this.lastPage = payload.currentNum;
          this.$emit('page-change', payload.currentNum);
        }
      });
      this.instance.on('backTop', () => this.$emit('back-top'));
    },
    destroy() {
      if (this.instance) {
        this.instance.destroy();
        this.instance = null;
      }
    },
    goToPage(n) { this.instance && this.instance.goToPage(n); },
    nextPage() { this.instance && this.instance.nextPage(); },
    prevPage() { this.instance && this.instance.prevPage(); },
    zoomIn() { this.instance && this.instance.zoomIn(); },
    zoomOut() { this.instance && this.instance.zoomOut(); },
    setScale(s) { this.instance && this.instance.setScale(s); },
    getStatus() { return this.instance ? this.instance.getStatus() : null; },
  },
};
</script>
```
